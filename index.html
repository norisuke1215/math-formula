💡 簡単にメールやLINE通知を受け取る方法 (GASコード例)

Google Apps Script (GAS) に貼り付けて「ウェブアプリ」としてデプロイしてください：

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var student = data.student || "名前なし";
  var answers = data.answers || [];
  
  var body = "【数学演習 提出通知】\n生徒: " + student + "\n\n";
  answers.forEach(function(item, idx) {
    body += "問" + (idx+1) + ": " + item.answer + "\n";
  });
  
  // メールで通知を受け取る場合
  MailApp.sendEmail("your-email@gmail.com", "数学演習提出: " + student, body);
  
  return ContentService.createTextOutput(JSON.stringify({status: "ok"}))
    .setMimeType(ContentService.MimeType.JSON);
}
